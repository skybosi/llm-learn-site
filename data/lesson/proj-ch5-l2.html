二、实现指导与代码骨架#

步骤 1：文档解析与切块（src/ingest.py）#

把 data/ 下的 PDF/Markdown 变成结构化"文档块"，并向量化入库。

# -*- coding: utf-8 -*-
# ingest.py —— 解析文档 → 切块 → 向量化 → 写入 Chroma
import os
import chromadb
from chromadb.utils import embedding_functions
from langchain_text_splitters import RecursiveCharacterTextSplitter
import fitz  # PyMuPDF，解析 PDF

DATA_DIR = "data"

def read_pdf(path: str) -> str:
    """读取一个 PDF 的全文（纯文本抽取）。"""
    doc = fitz.open(path)
    return "\n".join(page.get_text() for page in doc)

def read_text(path: str) -> str:
    """读取 txt/md 文件。"""
    with open(path, encoding="utf-8") as f:
        return f.read()

def load_documents() -> list[tuple[str, str]]:
    """返回 [(来源文件名, 文本内容), ...]。"""
    docs = []
    for name in os.listdir(DATA_DIR):
        p = os.path.join(DATA_DIR, name)
        if name.lower().endswith(".pdf"):
            docs.append((name, read_pdf(p)))
        elif name.lower().endswith((".txt", ".md")):
            docs.append((name, read_text(p)))
    return docs

def chunk_text(text: str, chunk_size: int = 512, overlap: int = 64) -> list[str]:
    """按字符滑窗切块：块 512 字符、重叠 64，保证上下文不丢。"""
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size, chunk_overlap=overlap,
        separators=["\n\n", "\n", "。", "！", "？", " ", ""],
    )
    return splitter.split_text(text)

def main():
    ef = embedding_functions.SentenceTransformerEmbeddingFunction(
        model_name="BAAI/bge-small-zh-v1.5"
    )
    client = chromadb.PersistentClient(path="./chroma_db")
    col = client.get_or_create_collection("docs", embedding_function=ef)

    ids, documents, metadatas = [], [], []
    for fname, text in load_documents():
        for i, chunk in enumerate(chunk_text(text)):
            ids.append(f"{fname}#{i}")
            documents.append(chunk)
            metadatas.append({"source": fname, "chunk": i})
    col.upsert(ids=ids, documents=documents, metadatas=metadatas)
    print(f"入库完成：共 {len(ids)} 个文档块")

if __name__ == "__main__":
    main()

步骤 2：检索（src/retrieve.py）#

根据问题，从向量库里召回 top-k 个最相关的文档块，并带上来源（RAG 的"引用"就是从这里来的）。

# -*- coding: utf-8 -*-
# retrieve.py —— 语义检索 top-k 相关块，返回内容 + 来源
import chromadb
from chromadb.utils import embedding_functions

def get_collection():
    ef = embedding_functions.SentenceTransformerEmbeddingFunction(
        model_name="BAAI/bge-small-zh-v1.5"
    )
    client = chromadb.PersistentClient(path="./chroma_db")
    return client.get_collection("docs", embedding_function=ef)

def retrieve(query: str, top_k: int = 4) -> list[dict]:
    """返回 [{"text":..., "source":...}, ...]，按相关度排序。"""
    col = get_collection()
    res = col.query(query_texts=[query], n_results=top_k)
    hits = []
    for doc, meta, dist in zip(res["documents"][0],
                               res["metadatas"][0],
                               res["distances"][0]):
        hits.append({"text": doc, "source": meta.get("source", "?"),
                     "distance": dist})
    return hits

步骤 3：RAG 生成（src/rag.py）#

把检索到的段落拼进 prompt，调用 P10 部署好的 vLLM 服务生成回答，并要求引用来源。

# -*- coding: utf-8 -*-
# rag.py —— 拼 prompt 调 LLM 生成带引用的回答
from openai import OpenAI
from retrieve import retrieve

# P10 部署的 vLLM 服务（OpenAI 兼容接口），复用前一章的成果
client = OpenAI(base_url="http://localhost:8000/v1", api_key="EMPTY")
MODEL = "Qwen/Qwen2.5-7B-Instruct"

SYSTEM = (
    "你是一个知识库问答助手。请只根据下面提供的资料回答问题；"
    "如果资料不足以回答，请明确说'资料中没有相关信息'，不要编造。"
    "回答末尾用 [来源：文件名] 标注你引用了哪些资料。"
)

def build_prompt(query: str, hits: list[dict]) -> str:
    context = "\n\n".join(
        f"[{h['source']}]\n{h['text']}" for h in hits
    )
    return f"资料：\n{context}\n\n问题：{query}\n回答："

def answer(query: str, top_k: int = 4) -> str:
    hits = retrieve(query, top_k=top_k)
    user = build_prompt(query, hits)
    resp = client.chat.completions.create(
        model=MODEL,
        messages=[{"role": "system", "content": SYSTEM},
                  {"role": "user", "content": user}],
        temperature=0.1,          # 低温度，追求稳定、少胡编
        max_tokens=256,
    )
    return resp.choices[0].message.content

if __name__ == "__main__":
    while True:
        q = input("问题（q 退出）：").strip()
        if q == "q":
            break
        print(answer(q))

步骤 4：服务化（src/serve.py）#

用 FastAPI 把 rag.answer 包成 HTTP 接口，返回答案 + 引用来源，前端可调用。

# -*- coding: utf-8 -*-
# serve.py —— FastAPI 服务：POST /ask 返回答案与引用
from fastapi import FastAPI
from pydantic import BaseModel
from rag import answer
from retrieve import retrieve

app = FastAPI(title="知识库助手")

class AskReq(BaseModel):
    question: str

@app.post("/ask")
def ask(req: AskReq):
    hits = retrieve(req.question, top_k=4)
    reply = answer(req.question)
    return {
        "question": req.question,
        "answer": reply,
        "sources": [{"source": h["source"], "snippet": h["text"][:80]}
                    for h in hits],
    }

# 启动：uvicorn serve:app --host 0.0.0.0 --port 9000

步骤 5：容器化 + 前端（上线）#

Dockerfile 把服务打成镜像；前端一个极简网页调用 /ask 展示问答与引用。

# Dockerfile —— 构建知识库助手服务镜像
FROM python:3.10-slim

WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .

# 默认暴露 FastAPI 端口
EXPOSE 9000
CMD ["uvicorn", "src.serve:app", "--host", "0.0.0.0", "--port", "9000"]
<!-- web/index.html —— 极简前端：输入问题，展示答案与来源 -->
<!DOCTYPE html>
<html lang="zh">
<head><meta charset="utf-8"><title>知识库助手</title></head>
<body>
  <h2>我的知识库助手</h2>
  <input id="q" placeholder="输入你的问题" style="width:80%">
  <button onclick="ask()">提问</button>
  <div id="answer"></div>
  <script>
    async function ask() {
      const q = document.getElementById("q").value;
      const res = await fetch("http://localhost:9000/ask", {
        method: "POST",
        headers: {"Content-Type": "application/json"},
        body: JSON.stringify({question: q}),
      });
      const data = await res.json();
      document.getElementById("answer").innerHTML =
        "<p><b>答：</b>" + data.answer + "</p>" +
        "<p><b>来源：</b>" + data.sources.map(s => s.source).join("、") + "</p>";
    }
  </script>
</body>
</html>

若目标平台是微信小程序，前端逻辑相同：小程序 wx.request 调 POST /ask，把 answer 与 sources 渲染到页面即可。后端接口不变，只是前端容器从浏览器换成小程序——这正是"服务化"带来的复用价值。

三、验收与复盘#

1. 本课验收（实现完成标志）#

  • python src/ingest.py 能把文档切块入库，打印出块数；
  • python src/rag.py 能交互问答，且回答带 [来源：文件名]；
  • uvicorn src.serve:app 起服务后，POST /ask 返回答案 + 来源；
  • 前端页面能输入问题并展示答案与来源（本地 9000 端口跑通）。

2. 动手练习#

  1. 把 top_k 从 4 改成 2、8，观察回答质量变化，说明"上下文太少 vs 噪声太多"的权衡。
  2. 给 SYSTEM 加上"先给出结论再给依据"的要求，观察回答结构是否更清晰——体会"系统提示词"对产品体验的影响。
  3. 记录一次完整请求的耗时（检索 + 生成），拆出"检索多少 ms、生成多少 ms"，作为后面性能优化的基线。

3. 复盘要点（实现阶段要避开的坑）#

  • 胡编（幻觉）：系统提示词里"资料没有就明说"是防幻觉的第一道闸，务必保留；温度调低（0.1~0.3）也能显著减少瞎编。
  • 引用丢失：来源必须在检索阶段就带出来（metadata 里存 source），生成阶段只能"引用"不能"编造来源"。
  • 接口不通：本地跑通 ≠ 上线能访问，跨域（CORS）与监听地址（0.0.0.0）是网页前端最容易踩的两个坑。
  • 生成服务没复用：别忘了 P10 部署的 vLLM 就是本项目的生成后端，复用它是"技能串联"的证明，面试里主动讲这一点。
